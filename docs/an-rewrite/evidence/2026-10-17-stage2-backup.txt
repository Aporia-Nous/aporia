// Stage-1 pre-migration backup of the DEV database.
// Checkpoints the WAL, then VACUUM INTO a .bak-stage1 marker next to the DB, then verifies it.
// Usage: bun backup-stage1.ts <source.db> <target.db>
import { Database } from "bun:sqlite"
import { existsSync, statSync } from "node:fs"

const [src, dst] = process.argv.slice(2) as [string, string]
if (!src || !dst) throw new Error("usage: backup-stage1.ts <source> <target>")
if (existsSync(dst)) throw new Error(`target already exists: ${dst}`)

// 1. Flush the WAL into the main file so the copy cannot depend on sidecar state.
{
  const db = new Database(src, { readwrite: true } as any)
  const mode = db.query("pragma journal_mode").get() as { journal_mode: string }
  console.log(`journal_mode = ${mode.journal_mode}`)
  if (mode.journal_mode === "wal") {
    const res = db.query("pragma wal_checkpoint(TRUNCATE)").get() as Record<string, number>
    console.log(`wal_checkpoint(TRUNCATE) = ${JSON.stringify(res)}`)
  }
  db.close()
}

// 2. Consistent copy.
{
  const db = new Database(src, { readonly: true })
  db.exec(`VACUUM INTO '${dst.replace(/\\/g, "/")}'`)
  db.close()
}

const size = (p: string) => `${(statSync(p).size / 1024 ** 3).toFixed(3)} GB`
console.log(`source ${size(src)} -> target ${size(dst)}`)

const check = new Database(dst, { readonly: true })
console.log(`integrity_check = ${JSON.stringify(check.query("pragma integrity_check").get())}`)
for (const t of ["session_v2", "session_message", "project", "migration"]) {
  const rows = check.query(`select count(*) c from ${t}`).get() as { c: number }
  console.log(`${t} rows = ${rows.c.toLocaleString("en-US")}`)
}
const tables = check.query("select count(*) c from sqlite_master where type='table' and name not like 'sqlite_%'").get() as { c: number }
console.log(`tables = ${tables.c}`)
check.close()
